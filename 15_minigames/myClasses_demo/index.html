// Ball er kassen øverst. Felterne i diagrammet sættes i constructoren.
class Ball {
  constructor(x, y, r, img, col, xSpeed, ySpeed, bounds){ 
    this.diam = r
    this.velocity = createVector(xSpeed || 0, ySpeed || 0)
    this.position = createVector(x, y)
    this.img = null
    this.col = null
    if(typeof img === 'string'){
      this.col = img
    } else if(img){
      this.img = img
    }
    if(col) this.col = col
    this.bounds = bounds || [false, false, false, false]
    this.mass = this.diam
    this.fixed = false
  }
  // update() lægger velocity oven i position
  update(){
    this.position.add(this.velocity)
  }
  show(){
    if(this.img){
      imageMode(CENTER)
      image(this.img, this.position.x, this.position.y, this.diam, this.diam)
    }else if(this.col){
      noStroke()
      fill(this.col)
      circle(this.position.x, this.position.y, this.diam)
    }
  }
  // constrain() vender farten, når bolden rammer en væg der er slået til
  constrain(){
    var r = this.diam / 2
    var b = this.bounds
    if(b[0] && this.position.x < r){
      this.position.x = r
      this.velocity.x *= -1
    }
    if(b[1] && this.position.x > width - r){
      this.position.x = width - r
      this.velocity.x *= -1
    }
    if(b[2] && this.position.y < r){
      this.position.y = r
      this.velocity.y *= -1
    }
    if(b[3] && this.position.y > height - r){
      this.position.y = height - r
      this.velocity.y *= -1
    }
  }
  // hit? er sand, når afstanden er mindre end de to radier lagt sammen
  hit(anotherBall){
    var b = anotherBall
    var totalR = (this.diam + b.diam) / 2
    var d = dist(this.position.x, this.position.y, b.position.x, b.position.y)
    return d <= totalR
  }
}

// Hul pil: JumpingBall extends Ball. player i Hop er en af de her.
class JumpingBall extends Ball{
  constructor(x, y, r, img, jump, col, bounds){
    super(x, y, r, img, col, 0, 0, bounds || [false, false, false, true])
    this.jumpForce = jump
  }
  // først tyngde og friktion, så Balls update som flytter positionen
  update(){
    this.velocity.add(gravity)
    this.velocity.y *= friction
    super.update()
  }
  // hop? ja. Trækker jumpForce fra velocity.y
  jump(){
    this.velocity.y -= this.jumpForce
  }
}

// Hul pil igen: Target extends Ball. Det er pointboldene.
class Target extends Ball {
  constructor(x, y, r, img, col, xSpeed, ySpeed, bounds, worth) {
    super(x, y, r, img, col, xSpeed, ySpeed, bounds)
    this.worth = worth
    this.taken = false
  }

  // n i flowchartet. 0 hvis den allerede er taget, ellers worth
  take() {
    if (this.taken) return 0
    this.taken = true
    return this.worth
  }

  // place(x, y) flytter den og gør den klar til at blive ramt igen
  place(x, y) {
    this.position.set(x, y)
    this.taken = false
  }

  show() {
    if (this.taken) return
    super.show()
    fill(0)
    noStroke()
    textAlign(CENTER, CENTER)
    textSize(this.diam * 0.28)
    text(this.worth, this.position.x, this.position.y)
  }
}

function hitTargets(hitter, after) {
    targets.map(function(t) {
        t.update()
        t.constrain()
        // hit? player.hit(t)
        if (hitter.hit(t)) {
            var n = t.take()
            // n? take() gav worth. 0 betyder allerede taget, så vi springer over
            if (n) {
                scorePoint(n)
                recycle(t)
                if (after) after()
            }
        }
        t.show()
    })
}
